if(WIN32)
    set(gui_process process_win32.c)
elseif(CMAKE_SYSTEM_NAME MATCHES "^(Linux|FreeBSD|Android)$")
    set(gui_process process_posix.c)
else()
    message(FATAL_ERROR "scarletnx-gui currently supports Linux, FreeBSD and Windows")
endif()
include("${PROJECT_SOURCE_DIR}/cmake/gui_themes.cmake")
add_library(snx_gui_support STATIC model.c browser.c steps.c theme.c recent.c editor.c ${gui_process}
            "${CMAKE_BINARY_DIR}/generated/gui_themes.c")
target_link_libraries(snx_gui_support PUBLIC scarletnx-common)
target_include_directories(snx_gui_support PRIVATE "${CMAKE_BINARY_DIR}/generated")
# The driver configures libgccjit in its own directory scope; its target is
# visible here even though SNX_GCCJIT_FOUND is not.
if(TARGET snx_driver_gccjit)
    target_compile_definitions(snx_gui_support PUBLIC SNX_GUI_HAS_GCCJIT=1)
else()
    target_compile_definitions(snx_gui_support PUBLIC SNX_GUI_HAS_GCCJIT=0)
endif()
target_include_directories(snx_gui_support PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(snx_gui_support PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_compile_options(snx_gui_support PRIVATE -Wall -Wextra -Werror=missing-prototypes)
if(WIN32)
    target_compile_definitions(snx_gui_support PRIVATE _WIN32_WINNT=0x0602 _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(snx_gui_support PUBLIC psapi)
endif()
include("${PROJECT_SOURCE_DIR}/cmake/window_icon.cmake")
include("${PROJECT_SOURCE_DIR}/cmake/build_info.cmake")
if(ANDROID)
    # the app's menu (dist/android): SDL's activity loads libmain.so and runs its SDL_main
    add_library(scarletnx-gui SHARED main.c android.c)
    set_target_properties(scarletnx-gui PROPERTIES OUTPUT_NAME main)
    install(TARGETS scarletnx-gui LIBRARY DESTINATION lib/${ANDROID_ABI})
    install(DIRECTORY "${SDL3_SOURCE_DIR}/android-project/app/src/main/java/" DESTINATION java)
else()
    add_executable(scarletnx-gui main.c)
endif()
set_target_properties(scarletnx-gui PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_options(scarletnx-gui PRIVATE -Wall -Wextra -Werror=missing-prototypes)
target_compile_definitions(scarletnx-gui PRIVATE SNX_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
target_link_libraries(scarletnx-gui PRIVATE snx_gui_support scarletnx-common SDL3::SDL3 snx_window_icon snx_build_info)
if(WIN32)
    enable_language(RC)
    target_sources(scarletnx-gui PRIVATE ../nx/runtime/platform/runtime.rc)
    target_include_directories(scarletnx-gui PRIVATE ../nx/runtime/platform)
    target_link_options(scarletnx-gui PRIVATE -mwindows -static)
endif()
include(GNUInstallDirs)
if(NOT ANDROID)
    install(TARGETS scarletnx-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(NOT WIN32)
    install(FILES "${PROJECT_SOURCE_DIR}/dist/scarletnx-gui.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications
            COMPONENT desktop)
endif()
if(SNX_SDL3_FROM_SOURCE AND NOT SNX_BUILD_NX)
    install(FILES "${SDL3_SOURCE_DIR}/LICENSE.txt" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/scarletnx/SDL3)
endif()
install(FILES FONT-NOTICE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/scarletnx)
# The built-in themes as files to copy and edit, beside the GUI where it looks for them, in the build and installed
file(GLOB_RECURSE gui_themes RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS themes/*/theme.ini)
foreach(theme IN LISTS gui_themes)
    configure_file(${theme} "${CMAKE_BINARY_DIR}/${theme}" COPYONLY)
endforeach()
install(DIRECTORY themes DESTINATION ${CMAKE_INSTALL_BINDIR})
